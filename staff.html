// ---------- ストップウォッチ ---------- // ボタン・キーの「押した瞬間」の時刻（event.timeStamp）で計るので、画面の描画の遅れは入らない let running = false, t0 = 0, raf = 0; let lastTrigger = -Infinity; // 直前にスタート/ストップした時刻 const MIN_GAP = 500; // この間隔（ミリ秒）より短い2回目の入力は無視する const held = new Set(); // 今押されっぱなしのキー function tick() { const sec = (performance.now() - t0) / 1000; const blind = Number($("blindSec").value) || 0; if (sec < blind) { $("disp").textContent = sec.toFixed(C.digits); $("disp").classList.remove("blind"); } else { $("disp").textContent = "??." + "?".repeat(C.digits); $("disp").classList.add("blind"); } raf = requestAnimationFrame(tick); } function trigger(ts) { if ($("entry").hidden) return; // ブラウザによってはイベントの時刻が別の基準で届くことがあるので、ずれていたら今の時刻を使う const now = performance.now(); if (!Number.isFinite(ts) || Math.abs(ts - now) > 1000) ts = now; // 1回の押下で入力が2回届いても、すぐ止まらないようにする if (ts - lastTrigger < MIN_GAP) return; lastTrigger = ts; if (!running) { running = true; t0 = ts; $("swBtn").textContent = "ストップ"; $("swBtn").classList.add("stop"); $("hint").className = "hint"; $("hint").textContent = C.target + "秒だと思ったら止める"; tick(); } else { running = false; cancelAnimationFrame(raf); const ms = quantize(ts - t0); $("disp").textContent = fmt(ms); $("disp").classList.remove("blind"); const d = diffOf(ms); $("hint").className = "hint" + (d === 0 ? " good" : ""); $("hint").textContent = d === 0 ? "ピッタリ！" : (ms < TARGET ? "早すぎ " : "遅すぎ ") + fmtDiff(d); $("time").value = fmt(ms); $("swBtn").textContent = "もう一度"; $("swBtn").classList.remove("stop"); if (!$("name").value) $("name").focus(); } } function resetWatch() { running = false; cancelAnimationFrame(raf); $("disp").textContent = ZERO; $("disp").classList.remove("blind"); $("hint").className = "hint"; $("hint").textContent = "スタートを押す（外部ボタンでもOK）"; $("swBtn").textContent = "スタート"; $("swBtn").classList.remove("stop"); } // 押した瞬間に反応させるため click ではなく pointerdown を使う $("swBtn").addEventListener("pointerdown", e => { e.preventDefault(); trigger(e.timeStamp); }); // キーを離したら「押しっぱなし」を解除 document.addEventListener("keyup", e => { held.delete(e.code || e.key); }); window.addEventListener("blur", () => held.clear()); document.addEventListener("keydown", e => { // 押しっぱなしで繰り返し届く入力は無視（repeat が付かない環境もあるので自前でも判定） const k = e.code || e.key; if (e.repeat || held.has(k)) { if (C.triggerKeys.includes(e.key) || C.triggerKeys.includes(e.code)) e.preventDefault(); return; } if (C.triggerKeys.includes(e.key) || C.triggerKeys.includes(e.code) || e.code === "Space") held.add(k); // 外部ボタン用のキー：どこにフォーカスがあっても反応 if (C.triggerKeys.includes(e.key) || C.triggerKeys.includes(e.code)) { e.preventDefault(); trigger(e.timeStamp); return; } // スペース：入力欄や他のボタンにいるときは普通の動作のまま if (e.code === "Space") { const t = e.target; if (t.tagName === "INPUT" || t.tagName === "TEXTAREA") return; if (t.tagName === "BUTTON" && t.id !== "swBtn") return; e.preventDefault(); trigger(e.timeStamp); } });
